#include <iostream>
using namespace std;

struct node {
    int value;
    node* next;
};

node* head = NULL;
node* tail = NULL;

// Insert First
void insertFirst(int n) {
    node *newnode = new node;
    newnode -> value = n;
    newnode -> next = NULL;
    if (head == NULL) {
        head = newnode;
        tail = newnode;
    } else {
        newnode -> next = head;
        head = newnode;
    }
}
// Delete First
void deleteFirst (){
    if (head== NULL) {
        cout<<"List Kosong!"<<endl;
        return;
    }
    node *temp = head;
    head = head -> next;
    if (head == NULL) tail == NULL;
    delete temp;
}

// Delete First Insert First - LIFO
void stacked(int n) {
    insertFirst(n);
}
void destacked() {
    deleteFirst();
}
void display() {
    node *temp = head;
    cout<<"List Linked List: \n";
    while (temp != NULL) {
        cout<<temp->value<<" -> ";
        temp = temp -> next;
    }
    cout<<endl;
}

int main() {
    system("clear");
    
    stacked(10); display();
    stacked(5); display();
    stacked(20); display();
    stacked(30); display();
    destacked(); display();
    stacked(90); display();

    return 0;
}
